বিশ্ববিদ্যালয়ের নিয়ম অনুযায়ী ক্রেডিট এবং উত্তরপত্রের পরিমাণের ওপর ভিত্তি করে ডাইনামিক রেট ক্যালকুলেট হয়:

// js/app.js: ক্যালকুলেশন রুল ইঞ্জিন
calculateStatutoryDutyAmount(dutyType, credit, studentCount, durationHours) {
    switch (dutyType) {
        case 'question_setting':
            // ৩.০০ ক্রেডিট বা তার বেশি হলে ৪০০০/- টাকা (হাফ পার্ট ২০০০/- টাকা)
            // ২.০০ ক্রেডিট হলে ২৪০০/- টাকা (হাফ পার্ট ১২০০/- টাকা)
            return credit >= 3.0 ? 4000 : (credit >= 2.0 ? 2400 : 1600);

        case 'moderation_board':
            // ৪ জন সদস্যের জন্য প্রতি কোর্সে ১৬০০/- টাকা
            return 1600;

        case 'script_evaluation':
            // প্রতি উত্তরপত্র ১২০/- টাকা (সর্বনিম্ন ১০০০/- টাকা)
            const scriptTotal = studentCount * 120;
            return Math.max(scriptTotal, 1000);

        case 'class_test':
            // ৩.০০ ক্রেডিটে ৪টি, ২.০০ ক্রেডিটে ৩টি, ১.০০ ক্রেডিটে ২টি সিটি (প্রতি সিটি ১২০০/- টাকা)
            const ctCount = credit >= 3.0 ? 4 : (credit >= 2.0 ? 3 : 2);
            return ctCount * 1200;

        case 'lab_exam':
            // ল্যাব এক্সাম ডিউটি প্রতি পরীক্ষা ১০০০/- টাকা
            return 1000;

        case 'tabulation':
            // ট্যাবুলেশন প্রতি শিক্ষার্থী ৭৫/- টাকা
            return studentCount * 75;

        default:
            return 0;
    }
}

// js/app.js: সংখ্যা থেকে বাংলা অংকে রূপান্তর
convertToBengaliNumerals(num) {
    if (num === null || num === undefined) return '০';
    const bnDigits = ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯'];
    return num.toString().replace(/[0-9]/g, d => bnDigits[parseInt(d, 10)]);
}

4. প্রজেক্টের কাজ (Core Purpose & Objective)

মাওলানা ভাসানী বিজ্ঞান ও প্রযুক্তি বিশ্ববিদ্যালয়ের ইনফরমেশন এন্ড কমিউনিকেশন টেকনোলজি (ICT) বিভাগের পরীক্ষা সংক্রান্ত যাবতীয় আর্থিক ও প্রশাসনিক জটিলতা নিরসন করাই এই প্রজেক্টের মূল কাজ।

  • কাগজবিহীন ডিজিটাল বিলিং: পরীক্ষার পর শিক্ষকদের পারিশ্রমিক হিসাব করার ধীরগতির হস্তলিখিত প্রক্রিয়াকে শতভাগ স্বয়ংক্রিয় করা।
  • হিউম্যান এরর নির্মূল: প্রশ্নপত্র তৈরি, মডারেশন, খাতা দেখা, ল্যাব পরীক্ষা এবং ক্লাস টেস্টের ম্যানুয়াল গণনার ভুল স্থায়ীভাবে বন্ধ করা।
  • অডিট ও অনুমোদন স্বচ্ছতা: বিভাগীয় চেয়ারম্যান, পরীক্ষা নিয়ন্ত্রক এবং অর্থ ও হিসাব দপ্তরের জন্য তাৎক্ষণিক অডিট লগ এবং সংবিধিবদ্ধ মডারেশন বিবরণী তৈরি করা।
  • ১:১ অফিসিয়াল ফর্ম রিপ্লিকা: বিশ্ববিদ্যালয়ের অনুমোদিত নীল-বর্ডারের ১৪-কলামের বিল ফরমের অবিকল লেআউটে তাৎক্ষণিক প্রিন্টেবল স্লিপ প্রদান করা।

  1. সময় সাশ্রয় (Time Efficiency): পূর্বে একটি সেমিস্টারের সকল শিক্ষকের বিল হাতে লিখতে ও হিসাব মেলাতে ২ থেকে ৩ সপ্তাহ সময় লাগত; বর্তমানে মাত্র ২ মিনিটে সমগ্র সেমিস্টারের বিল ও বিবরণী প্রস্তুত হয়।
  2. শতভাগ নির্ভুল গাণিতিক হিসাব (Zero Mathematical Error): জাভাস্ক্রিপ্ট রুল ইঞ্জিনের কারণে ভুল রেট প্রয়োগ বা যোগের ভুল হওয়ার কোনো অবকাশ নেই।
  3. আর্থিক অডিট জবাবদিহিতা (Audit Trail & Transparency): প্রতিটি বিলের সাথে সংশ্লিষ্ট কোর্স কোড, সেমিস্টার, ক্রেডিট ও তারিখ স্বয়ংক্রিয়ভাবে সংরক্ষিত থাকে।
  4. জিরো সার্ভার খরচ (Zero Operational Cost): কোনো পেইড ক্লাউড সার্ভার বা ডাটাবেস হোস্টিং ফি প্রয়োজন হয় না।
  5. কাগজ অপচয় হ্রাস: স্ক্রিনে ফাইনাল চেক করার পর কেবল অনুমোদিত বিলটি প্রিন্ট করা যায়, ফলে ভুল বিল রিপ্রিন্ট করার কাগজ অপচয় রোধ হয়।

7. সীমাবদ্ধতা (Current Limitations)

  • ব্রাউজার লোকাল স্টোরেজ নির্ভরতা: ডাটা ব্যবহারকারীর লোকাল ব্রাউজারের localStorage-এ জমা থাকে। ব্রাউজার ক্যাশ ও ডাটা সম্পূর্ণ মুছে ফেললে ব্যাকআপ ফাইল থেকে রিস্টোর করতে হয়।
  • একক ব্যবহারকারী মোড (Single-user Local Environment): একাধিক শিক্ষক একই সাথে বিভিন্ন কম্পিউটার থেকে রিয়েল-টাইমে ডাটা এন্ট্রি দিতে পারেন না (কেননা সেন্ট্রাল ক্লাউড ডাটাবেজ এখনও সংযুক্ত হয়নি)।
  • ডিজিটাল ক্রিপ্টোগ্রাফিক স্বাক্ষরের অনুপস্থিতি: প্রিন্ট করার পর বিভাগীয় প্রধান ও শিক্ষকদের সনাতন কলমে স্বাক্ষর করতে হয়; পিকেআই (PKI) ভিত্তিক ডিজিটাল সিগনেচার এখনও যুক্ত করা হয়নি।

8. ভবিষ্যৎ উন্নয়ন পরিকল্পনা (Future Enhancements)

  • ক্লাউড ডাটাবেস ইন্টিগ্রেশন: Firebase বা PostgreSQL ব্যাকএন্ড যুক্ত করে মাল্টি-ইউজার সিঙ্ক্রোনাইজেশন।
  • রোল-ভিত্তিক অ্যাক্সেস কন্ট্রোল (RBAC): শিক্ষক, পরীক্ষা কমিটি সভাপতি, বিভাগীয় চেয়ারম্যান এবং অডিট সেকশনের পৃথক লগইন পোর্টাল।
  • এসএমএস ও ইমেইল নোটিফিকেশন: বিল অনুমোদন ও অ্যাকাউন্টস সেকশন থেকে বিল ছাড় হলে শিক্ষকদের মোবাইলে স্বয়ংক্রিয় অ্যালার্ট।
  • ডিজিটাল সিগনেচার: চেয়ারম্যান ও শিক্ষকদের ওটিপি/ডিজিটাল সাইন দিয়ে সরাসরি ই-সাইন বিল অনুমোদন।
